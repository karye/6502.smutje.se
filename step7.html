Steg 7 — I/O-expansion med W65C22 VIA och LCD

Mål

Koppla in en W65C22 VIA (Versatile Interface Adapter) som ger 20 I/O-linjer och styr en parallell LCD-display direkt från 6502:ans buss. En 74HC00 används för adressavkodning så att VIA:n och övriga enheter får rätt chip select-signaler. Nu är vi på väg mot en riktig dator — LCD:n styrs av CPU:n, inte av Arduino.

Kopplingsschema

Adressavkodning med 74HC00

74HC00:an har fyra NAND-grindar. Vi använder en av dem för att avkoda adresslinjerna:

  • Grind B (pins 4,5,6): A14 (pin 4) NAND A15 (pin 5) → /CS_VIA (pin 6)

Resultat: VIA:n aktiveras (CS1=HÖG, /CS2=LÅG) när både A14 och A15 är höga, dvs. vid adress $C000–$FFFF. Registerväljarna RS0–RS3 kopplas till A0–A3, vilket ger 16 register på $C000–$C00F.

VIAns port A används för LCD:ns kontrollsignaler (RS, R/W, E). Port B hanterar 8-bitars databussen. CPU:n sätter PA1=0 (write) vid alla LCD-accesser — R/W-linjen är alltså alltid låg i praktiken.

Arduino-kod — komplett main.cpp

När VIA:n är inkopplad ska Arduinon inte svara på adresser i VIA:ns område ($C000–$C00F) — den fysiska VIA-kretsen hanterar dessa. Vid läsning från VIA-adresser sätts PORTA till INPUT så att VIA:n kan driva databussen. LCD-biblioteket är borttaget — LCD:n styrs nu helt av 6502 via VIA.

#include <Arduino.h>

// -----------------------------------------------------------
// Steg 7 — VIA + LCD (LCD styrs av 6502 via W65C22 VIA)
// -----------------------------------------------------------

#define PHI2     2   // CPU pin 37
#define RESB     4   // CPU pin 40
#define RWB      3   // CPU pin 34
#define SYNC     13  // CPU pin 7 → Arduino D13
#define BTN_CLK  11  // Knapp: en klockcykel per tryck
#define BTN_INSTR 12 // Knapp: kör till nästa instruktion

uint8_t ram[1024];     // $0000–$03FF (stack + data + program)
uint8_t vectors[6];     // $FFFA–$FFFF

// VIA-adresser — VIA:n är en fysisk krets, Arduinon ska INTE svara här
#define VIA_BASE 0xC000

bool is_via(uint16_t addr) {
  return (addr >= VIA_BASE && addr < VIA_BASE + 16);
}

uint8_t read_mem(uint16_t addr) {
  if (is_via(addr)) return 0;  // VIA hanterar detta själv
  if (addr >= 0xFFFA && addr <= 0xFFFF)
    return vectors[addr - 0xFFFA];
  if (addr < 1024) return ram[addr];
  return 0xEA;
}

void write_mem(uint16_t addr, uint8_t val) {
  if (is_via(addr)) return;    // VIA hanterar detta själv
  if (addr >= 0xFFFA && addr <= 0xFFFF) {
    vectors[addr - 0xFFFA] = val; return;
  }
  if (addr < 1024) ram[addr] = val;
}

void pulse();  // Forward

void setup() {
  // --- Knappar och SYNC ---
  pinMode(SYNC, INPUT);
  pinMode(BTN_CLK,   INPUT_PULLUP);
  pinMode(BTN_INSTR, INPUT_PULLUP);

  // --- Håll CPU i reset från start ---
  pinMode(RESB, OUTPUT); digitalWrite(RESB, LOW);
  pinMode(PHI2, OUTPUT); digitalWrite(PHI2, LOW);
  Serial.begin(115200);
  pinMode(RWB, INPUT);

  DDRK = 0x00;  // A8–A15
  DDRF = 0x00;  // A0–A7
  DDRA = 0x00;

  // --- Reset-vektor → $8000 ---
  write_mem(0xFFFC, 0x00);
  write_mem(0xFFFD, 0x80);

  // LCD-init via VIA (6502-program på $8000):
  write_mem(0x8000, 0xA9); write_mem(0x8001, 0xFF);  // LDA #$FF
  write_mem(0x8002, 0x8D); write_mem(0x8003, 0x02); write_mem(0x8004, 0xC0);  // STA $C002 (DDRB)
  write_mem(0x8005, 0x8D); write_mem(0x8006, 0x03); write_mem(0x8007, 0xC0);  // STA $C003 (DDRA)
  write_mem(0x8008, 0xA9); write_mem(0x8009, 0x38);  // LDA #$38 (8-bit, 2 rader)
  write_mem(0x800A, 0x8D); write_mem(0x800B, 0x00); write_mem(0x800C, 0xC0);  // STA $C000 (PORTB)
  write_mem(0x800D, 0xA9); write_mem(0x800E, 0x04);  // LDA #$04 (E=1)
  write_mem(0x800F, 0x8D); write_mem(0x8010, 0x01); write_mem(0x8011, 0xC0);  // STA $C001 (PORTA)
  write_mem(0x8012, 0xA9); write_mem(0x8013, 0x00);  // LDA #$00 (E=0)
  write_mem(0x8014, 0x8D); write_mem(0x8015, 0x01); write_mem(0x8016, 0xC0);  // STA $C001
  write_mem(0x8017, 0xA9); write_mem(0x8018, 0x0C);  // LDA #$0C (Display ON)
  write_mem(0x8019, 0x8D); write_mem(0x801A, 0x00); write_mem(0x801B, 0xC0);  // STA $C000
  write_mem(0x801C, 0xA9); write_mem(0x801D, 0x04);  // LDA #$04 (E=1)
  write_mem(0x801E, 0x8D); write_mem(0x801F, 0x01); write_mem(0x8020, 0xC0);  // STA $C001
  write_mem(0x8021, 0xA9); write_mem(0x8022, 0x00);  // LDA #$00 (E=0)
  write_mem(0x8023, 0x8D); write_mem(0x8024, 0x01); write_mem(0x8025, 0xC0);  // STA $C001
  write_mem(0x8026, 0x4C); write_mem(0x8027, 0x26); write_mem(0x8028, 0x80);  // JMP $8026 (loop)

  // --- Reset-sekvens ---
  for (int i = 0; i < 5; i++) pulse();
  digitalWrite(RESB, HIGH);
  for (int i = 0; i < 40; i++) pulse();  // Fler cykler för LCD-init
}

void pulse() {
  digitalWrite(PHI2, LOW);
  delay(25);

  uint16_t a = (PINF << 0) | (PINK << 8);
  bool rw = digitalRead(RWB);

  if (rw) {
    if (is_via(a)) {
      DDRA = 0x00;  // VIA driver databussen — Arduino håller sig undan
    } else {
      DDRA = 0xFF;
      PORTA = read_mem(a);
    }
  } else {
    DDRA = 0x00;    // CPU driver databussen vid skrivning
  }

  digitalWrite(PHI2, HIGH);
  delay(25);

  if (!rw && !is_via(a)) write_mem(a, PINA);

  Serial.print(rw ? "R" : "W");
  Serial.print(" $");
  Serial.println(a, HEX);
}

void loop() {
  // --- Knapp 1: En klockcykel ---
  if (!digitalRead(BTN_CLK)) {
    delay(50);
    while (!digitalRead(BTN_CLK));
    delay(50);
    pulse();
  }

  // --- Knapp 2: Kör till nästa instruktion ---
  if (!digitalRead(BTN_INSTR)) {
    delay(50);
    while (!digitalRead(BTN_INSTR));
    delay(50);
    do {
      pulse();
    } while (!digitalRead(SYNC));
  }
}

6502-programmet — LCD-initiering

Ovanstående Arduino-kod laddar automatiskt detta 6502-program vid $8000. Programmet sätter VIA:ns portar som utgångar, initierar LCD:n i 8-bitarsläge och slår på displayen.

Testning

  1. Koppla in W65C22, 74HC00 och LCD enligt schemat
  2. Verifiera att VIA:n svarar på adress $C000–$C00F via serieövervakaren
  3. Skicka ett init-kommando till LCD:n via VIA — displayen ska visa svarta rutor på första raden (tecken på att den är i 8-bitarsläge)
  4. Skriv "Hello 6502!" till LCD:n via VIA:ns port B
✅ Förväntat: LCD:n initieras och visar text — helt styrd av 6502 via VIA:n. Arduinon är nu bara minnesemulator och klocka.

Systemarkitektur efter steg 7

6502 CPU ─── Adressbuss (A0–A15) ───┬── Arduino (ROM $8000–$BFFF)
            │                          ├── 74HC00 → VIA /CS2
            ├── Databuss (D0–D7) ──────┤
            ├── R/W ───────────────────┤
            ├── PHI2 ──────────────────┤
            └── RESB ──────────────────┘

W65C22 VIA ─── PA0–PA2 ──→ LCD (RS, R/W, E)
            ─── PB0–PB7 ──→ LCD (DB0–DB7)
            ─── /CS2 ← 74HC00 (A14 OR A15)
            ─── CS1 ← +5V
            ─── RS0–RS3 ← A0–A3

VIAns register är minnesmappade på $C000–$C00F. CPU:n läser/skriver till VIA:n precis som till vanligt RAM — men istället för att lagra data styr den I/O-pinnar.
